# Fedory GRUB theme.
#
# Colours are the Fedory Vesper palette (themes/fedory-vesper/colors.toml), so
# the boot menu, the Plymouth splash, and the SDDM greeter read as one system.
#
# Deliberately pixmap-free: every element is drawn with plain colours rather
# than sliced *_c.png/_w.png sets. Pixmap styles are what make most GRUB themes
# brittle -- a missing slice silently drops the widget -- and nothing here needs
# them. The only image is the logo.
#
# fedory.pf2 is generated at install time by bin/fedory-refresh-grub and lives
# in this directory on the installed system. /etc/grub.d/00_header scans the
# theme directory and emits a `loadfont` for every .pf2 it finds, which is why
# the font has to sit here rather than in /boot/grub2/fonts.

title-text: ""
desktop-color: "#120C1C"
terminal-font: "Fedory Regular 16"
terminal-left: "8%"
terminal-top: "8%"
terminal-width: "84%"
terminal-height: "84%"
terminal-border: "0"

+ image {
    id = "__logo__"
    left = 50%-200
    top = 16%
    width = 400
    height = 94
    file = "logo.png"
}

+ boot_menu {
    left = 25%
    top = 38%
    width = 50%
    height = 38%

    item_font = "Fedory Regular 16"
    item_color = "#8E819B"
    selected_item_color = "#EDE6F4"
    item_height = 30
    item_padding = 6
    item_spacing = 6

    menu_pixmap_style = ""
    selected_item_pixmap_style = ""

    scrollbar = true
    scrollbar_width = 4
    scrollbar_thumb_color = "#3C2857"
}

+ progress_bar {
    id = "__timeout__"
    left = 25%
    top = 84%
    width = 50%
    height = 12

    font = "Fedory Regular 16"
    text_color = "#746584"
    text = "@TIMEOUT_NOTIFICATION_SHORT@"

    fg_color = "#B78CFF"
    bg_color = "#21162F"
    border_color = "#3C2857"
}
